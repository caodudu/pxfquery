可以函数化添加到bashrc里
```shell
# Function to start SOCKS5 proxy, test connectivity, and clean up after
socks5_proxy() {
  local port="${1:-1080}"  # Default to 1080 if no port is provided

  # Start SSH tunnel with SOCKS5 proxy
  echo "Starting SOCKS5 proxy on port $port..."
  ssh -f -N -D 127.0.0.1:$port caojun@login1

  # Set the necessary proxy environment variables
  export ALL_PROXY="socks5h://127.0.0.1:$port"
  export http_proxy="socks5h://127.0.0.1:$port"
  export https_proxy="socks5h://127.0.0.1:$port"
  export no_proxy="localhost,127.0.0.1,::1"  # Avoid local loopback

  echo "SOCKS5 proxy set up on port $port."

  # Test connectivity by downloading a test file
  local test_file="index.html"
  echo "Testing connectivity by downloading a test file..."

  # Attempt to download the file
  if curl -O https://www.baidu.com/index.html -o "$test_file"; then
    # Check if the file exists and is non-zero in size
    if [[ -f "$test_file" && -s "$test_file" ]]; then
      echo "Connectivity test successful. File downloaded."
      # Clean up the test file after download
      rm -f "$test_file"
      echo "Test file downloaded and deleted."
    else
      echo "Connectivity test failed. File not downloaded or is empty."
    fi
  else
    echo "Connectivity test failed. Download attempt unsuccessful."
  fi
}
```

## 1.2 socks5转http
有些软件不支持socks5，需要在计算节点里再转成http
比如opencode


1. 安装ghost
```
/public/home/caojun/software/internet/gost/gost
```


```shell
# SOCKS5 to HTTP proxy for OpenCode using gost
cat >> ~/.bashrc << 'EOF'

# SOCKS5 to HTTP proxy for OpenCode using gost
socks5_to_http_proxy() {
  local socks_port="${1:-1080}"
  local http_port="${2:-8118}"
  local gost_path="/public/home/caojun/software/internet/gost/gost"

  echo "========================================="
  echo "Setting up proxy chain for OpenCode"
  echo "========================================="

  # Check if SSH tunnel already exists
  if pgrep -f "ssh.*-D.*$socks_port" > /dev/null; then
    echo "✓ SOCKS5 tunnel already running on port $socks_port"
  else
    echo "Starting SOCKS5 proxy on port $socks_port..."
    ssh -f -N -D 127.0.0.1:$socks_port caojun@login1
    sleep 1
    if pgrep -f "ssh.*-D.*$socks_port" > /dev/null; then
      echo "✓ SOCKS5 tunnel started"
    else
      echo "✗ Failed to start SOCKS5 tunnel"
      return 1
    fi
  fi

  # Check if gost already exists
  if pgrep -f "gost.*$http_port" > /dev/null; then
    echo "✓ HTTP proxy already running on port $http_port"
  else
    echo "Starting HTTP proxy conversion on port $http_port..."
    $gost_path -L http://:$http_port -F socks5://127.0.0.1:$socks_port > /dev/null 2>&1 &
    local gost_pid=$!
    sleep 2
    if ps -p $gost_pid > /dev/null; then
      echo "✓ HTTP proxy started (PID: $gost_pid)"
    else
      echo "✗ Failed to start HTTP proxy"
      return 1
    fi
  fi

  # Set environment variables
  export http_proxy="http://127.0.0.1:$http_port"
  export https_proxy="http://127.0.0.1:$http_port"
  export ALL_PROXY="http://127.0.0.1:$http_port"
  export no_proxy="localhost,127.0.0.1,::1"

  echo ""
  echo "Proxy configuration:"
  echo "  SOCKS5: 127.0.0.1:$socks_port (SSH tunnel)"
  echo "  HTTP:   127.0.0.1:$http_port (gost)"
  echo ""

  # Test connectivity
  echo "Testing connectivity..."
  if curl -s -m 5 -I https://www.baidu.com > /dev/null 2>&1; then
    echo "✓ Connectivity test PASSED"
    echo ""
    echo "========================================="
    echo "Ready to use OpenCode!"
    echo "Run: opencode"
    echo "========================================="
  else
    echo "✗ Connectivity test FAILED"
    echo "Check your network setup"
    return 1
  fi
}

# Function to stop proxy
stop_proxy() {
  local socks_port="${1:-1080}"
  local http_port="${2:-8118}"

  echo "Stopping proxy services..."
  
  # Stop gost
  pkill -f "gost.*$http_port"
  
  # Stop SSH tunnel
  pkill -f "ssh.*-D.*$socks_port"
  
  # Unset environment variables
  unset http_proxy
  unset https_proxy
  unset ALL_PROXY
  unset no_proxy
  
  echo "✓ Proxy stopped and environment cleaned"
}

# Alias for convenience
alias start_proxy='socks5_to_http_proxy'

EOF
```



```shell
# 启动代理（使用默认端口 1080 和 8118）
socks5_to_http_proxy

# 或者使用别名
start_proxy

# 或者指定端口
socks5_to_http_proxy 1080 8118

# 启动 OpenCode
opencode

# 使用完后关闭代理
stop_proxy
```
