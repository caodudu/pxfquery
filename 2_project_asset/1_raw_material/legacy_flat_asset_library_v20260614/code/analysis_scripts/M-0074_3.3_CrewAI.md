## 3.2
```python
from crewai import Agent, Task, Crew

researcher = Agent(role='研究员', goal='搜索信息', backstory='生物专家', llm=llm)
analyst = Agent(role='分析师', goal='分析数据', backstory='数据专家', llm=llm, tools=tools)

task1 = Task(description='搜索蛋白质X的序列', agent=researcher)
task2 = Task(description='分析上个结果的长度', agent=analyst)

crew = Crew(agents=[researcher, analyst], tasks=[task1, task2])
result = crew.kickoff()
print(result)
```
